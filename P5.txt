import java.text.DecimalFormat;
import java.util.Arrays;
import java.util.Calendar;
import java.util.LinkedList;
import java.util.List;

import org.cloudbus.cloudsim.*;
import org.cloudbus.cloudsim.core.CloudSim;
import org.cloudbus.cloudsim.provisioners.*;

public class CloudSimExample2Shortcut {

    public static void main(String[] args) {
        Log.printLine("Starting CloudSimExample2...");

        try {
            // 1. Initialize CloudSim
            CloudSim.init(1, Calendar.getInstance(), false);

            // 2 & 3. Create Datacenter and Broker
            createDatacenter("Datacenter_0"); 
            DatacenterBroker broker = new DatacenterBroker("Broker");
            int brokerId = broker.getId();

            // 4. Create two VMs
            Vm vm1 = new Vm(0, brokerId, 250, 1, 512, 1000, 10000, "Xen", new CloudletSchedulerTimeShared());
            Vm vm2 = new Vm(1, brokerId, 250, 1, 512, 1000, 10000, "Xen", new CloudletSchedulerTimeShared());
            broker.submitVmList(Arrays.asList(vm1, vm2));

            // 5. Create two Cloudlets and bind them to specific VMs
            UtilizationModel um = new UtilizationModelFull();
            
            Cloudlet cloudlet1 = new Cloudlet(0, 250000, 1, 300, 300, um, um, um);
            cloudlet1.setUserId(brokerId);
            
            Cloudlet cloudlet2 = new Cloudlet(1, 250000, 1, 300, 300, um, um, um);
            cloudlet2.setUserId(brokerId);
            
            broker.submitCloudletList(Arrays.asList(cloudlet1, cloudlet2));

            // Bind the cloudlets to the VMs
            broker.bindCloudletToVm(cloudlet1.getCloudletId(), vm1.getId());
            broker.bindCloudletToVm(cloudlet2.getCloudletId(), vm2.getId());

            // 6. Run Simulation
            CloudSim.startSimulation();
            List<Cloudlet> newList = broker.getCloudletReceivedList();
            CloudSim.stopSimulation();

            // 7. Print Results
            printCloudletList(newList);

            Log.printLine("CloudSimExample2 finished!");
        } catch (Exception e) {
            e.printStackTrace();
            Log.printLine("The simulation has been terminated due to an unexpected error");
        }
    }

    private static Datacenter createDatacenter(String name) throws Exception {
        List<Pe> peList = Arrays.asList(new Pe(0, new PeProvisionerSimple(1000)));
        List<Host> hostList = Arrays.asList(new Host(
                0, new RamProvisionerSimple(2048), new BwProvisionerSimple(10000), 1000000, 
                peList, new VmSchedulerTimeShared(peList)
        ));

        DatacenterCharacteristics characteristics = new DatacenterCharacteristics(
                "x86", "Linux", "Xen", hostList, 10.0, 3.0, 0.05, 0.001, 0.0
        );
        
        // THE FIX: Added <Storage> inside the LinkedList diamond operator for Java 7 compatibility
        return new Datacenter(name, characteristics, new VmAllocationPolicySimple(hostList), new LinkedList<Storage>(), 0);
    }

    private static void printCloudletList(List<Cloudlet> list) {
        Log.printLine("\n========== OUTPUT ==========");
        Log.printLine("Cloudlet ID    STATUS    Data center ID    VM ID    Time    Start Time    Finish Time");

        DecimalFormat dft = new DecimalFormat("###.##");
        for (Cloudlet cloudlet : list) {
            if (cloudlet.getCloudletStatus() == Cloudlet.SUCCESS) {
                System.out.printf("    %-10d SUCCESS   %-17d %-8d %-7s %-13s %s%n",
                        cloudlet.getCloudletId(),
                        cloudlet.getResourceId(),
                        cloudlet.getVmId(),
                        dft.format(cloudlet.getActualCPUTime()),
                        dft.format(cloudlet.getExecStartTime()),
                        dft.format(cloudlet.getFinishTime())
                );
            }
        }
    }
}
